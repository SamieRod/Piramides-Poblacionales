#Correccion. Metodo Myers
#recuerda este solo es un diagnostico
#Para el metodo myers ya usamos la base prorateada que ya tiene los na agregados en la poblacion
#vamos a crear un residuo con una variable digit usando %% es un modulo para 
#%% significa modulo
#creamos dos variables nuevas con mutate que son: weights t la poblacion total
#usamos left join para unir dos data frames x, y. En este caso x suma desde 10 e y suma desde 20 
myers_tab <- left_join(
  base_mx_pror %>% 
    filter(age %in% c(10:80)) %>% 
    mutate(digit = age %% 10) %>% 
    group_by(digit) %>% 
    summarise(step1 = sum(pob), .groups = "drop") %>% 
    mutate(
      weights = c(1:10),
      tot_pop1 = step1 * weights
    ),
  base_mx_pror %>% 
    filter(age %in% c(10:80)) %>% 
    mutate(digit = age %% 10) %>% 
    filter(!age %in% c(10:19)) %>% 
    group_by(digit) %>% 
    summarise(step2 = sum(pob), .groups = "drop") %>% 
    mutate(
      weights = c(9:0),
      tot_pop2 = step2 * weights),
  by = "digit"
) %>% 
  select(digit, tot_pop1, tot_pop2) %>% 
  mutate(
    blended = tot_pop1 + tot_pop2,
    blend_perc = 100 * blended / sum(blended),
    deviat = abs(blend_perc - 10)
  )

#el valor proporcional lo multiplicamos por 100 para que nos mueste el porcentaje 

#el indice se divide entre 2 para que este entre 0 y 90
#por tanto si el resultado es mayor a 45 podemos inferir que es un mal estimador. en otras palabras hay un sesgo de las edades 
myers_tab %>% 
  summarise(myers = sum(deviat) / 2)
